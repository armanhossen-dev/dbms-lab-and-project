9. show product id and total_quantity sold only for products where total quantity is greater than 2

    select p_id, sum(quantity) as total_quantity from orders 
    group by p_id 
    having sum(quantity) > 2;

10.show the sold products name which has maximum total quantity of order

    select * 
    from  products 
    where p_id = (
        select p_id
        from orders
        group by p_id
        order by sum(quantity) DESC
        limit 1);

        +------+--------+-------+
        | p_id | p_name | price |
        +------+--------+-------+
        |  102 | Phone  | 40000 |
        +------+--------+-------+


11.show the products which product sales maximum times (not the highest order, just which product purchase by maximum customers)

    select * from products where p_id = ( select p_id from orders
    group by p_id  order by count(p_id)  desc limit 1);

    +------+--------+-------+
    | p_id | p_name | price |
    +------+--------+-------+
    |  102 | Phone  | 40000 |
    +------+--------+-------+


Extra: show with the quantity 

    select p.*, sum(o.quantity)  as total_sold
    from products p
    join orders   o
    on p.p_id = o.p_id
    GROUP BY p.p_id
    ORDER BY total_sold desc limit 1;

    +------+--------+-------+------------+
    | p_id | p_name | price | total_sold |
    +------+--------+-------+------------+
    |  102 | Phone  | 40000 |          4 |
    +------+--------+-------+------------+


12. show each customers bill (total cost and product names)

    SELECT 
    c.c_id, 
    c.c_name,
    GROUP_CONCAT(CONCAT(p.p_name, ' x', o.quantity) SEPARATOR ', ') AS products_bought,
    SUM(o.quantity * p.price) AS total_bill
    FROM customers c
    JOIN orders o ON c.c_id = o.c_id
    JOIN products p ON o.p_id = p.p_id
    GROUP BY c.c_id, c.c_name;
    
    +------+--------+-----------------------------------+------------+
    | c_id | c_name | products_bought                   | total_bill |
    +------+--------+-----------------------------------+------------+
    |    1 | Rahim  | Laptop x1, Headphone x3, Phone x2 |     175000 |
    |    2 | Karina | Phone x2                          |      80000 |
    |    3 | Imran  | Table x1                          |      30000 |
    |    4 | Tania  | Monitor x2                        |      40000 |
    +------+--------+-----------------------------------+------------+
