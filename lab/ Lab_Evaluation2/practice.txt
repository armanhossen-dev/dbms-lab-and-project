1.Task 1 — Procedure: getCustomersWithoutPurchases
DELIMITER $$

CREATE PROCEDURE getCustomersWithoutPurchases()
BEGIN
    SELECT Name
    FROM Customer
    WHERE CustomerID NOT IN (
        SELECT DISTINCT CustomerID FROM Purchase
    );
END$$

DELIMITER ;

-- Call it:
CALL getCustomersWithoutPurchases();


2.Task 2 — Procedure: updatePurchaseQuantity

DELIMITER $$

CREATE PROCEDURE updatePurchaseQuantity(
    IN purchaseID INT,
    IN newQuantity INT,
    OUT customerID INT
)
BEGIN
    UPDATE Purchase
    SET Quantity = newQuantity
    WHERE PurchaseID = purchaseID;

    SELECT CustomerID INTO customerID
    FROM Purchase
    WHERE PurchaseID = purchaseID;
END$$

DELIMITER ;

-- Call it:
CALL updatePurchaseQuantity(1, 5, @custID);
SELECT @custID;

3.Task 3 — Trigger: Log changes to PurchaseLog

DELIMITER $$

CREATE TRIGGER trg_LogQuantityChange
AFTER UPDATE ON Purchase
FOR EACH ROW
BEGIN
    IF OLD.Quantity <> NEW.Quantity THEN
        INSERT INTO PurchaseLog (PurchaseID, OldQuantity, NewQuantity, UpdatedAt)
        VALUES (OLD.PurchaseID, OLD.Quantity, NEW.Quantity, NOW());
    END IF;
END$$

DELIMITER ;
